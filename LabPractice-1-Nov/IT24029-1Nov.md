    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter n: ");

        if (!scanner.hasNextInt()) {
            System.out.println("Please enter a valid integer.");
            scanner.close();
            return;
        }

        int n = scanner.nextInt();

        if (n <= 0) {
            System.out.println("Please enter a positive integer.");
            scanner.close();
            return;
        }

        // Find nth prime number
        long prime = PrimeUtil.findNthPrime(n);

        // Display result
        System.out.println(
                "The " + n + "th prime number is: " + prime
        );

        // Save result to database
        try {

            PrimeDAO.savePrime(n, prime);

        } catch (SQLException e) {

            System.out.println("Database error: " + e.getMessage());

        }

        scanner.close();
    }
}

package com.prime;


import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.SQLException;

public class PrimeDAO {

    public static void savePrime(int n, long prime)
            throws SQLException {

        String sql = "INSERT INTO prime_results (input_n, prime_number) VALUES (?, ?)";

        try (
                Connection connection = DBConnection.getConnection();

                PreparedStatement statement =
                        connection.prepareStatement(sql)
        ) {

            statement.setInt(1, n);
            statement.setLong(2, prime);

            int rows = statement.executeUpdate();

            if (rows > 0) {
                System.out.println("Successfully saved to database!");
            }
        }
    }
}



package com.prime;

public class PrimeUtil {

    // Check whether a number is prime
    public static boolean isPrime(long number) {

        if (number < 2) {
            return false;
        }

        for (long i = 2; i <= number / i; i++) {

            if (number % i == 0) {
                return false;
            }
        }

        return true;
    }

    // Find the nth prime number
    public static long findNthPrime(int n) {

        int count = 0;
        long number = 1;

        while (count < n) {

            number++;

            if (isPrime(number)) {
                count++;
            }
        }

        return number;
    }
}

HelloServlet Code:
package com.prime;




import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;

@WebServlet("/check-prime")
public class HelloServlet extends HttpServlet {

    // Prime number checking method
    private boolean isPrime(int number) {

        if (number <= 1) {
            return false;
        }

        for (int i = 2; i <= Math.sqrt(number); i++) {

            if (number % i == 0) {
                return false;
            }
        }

        return true;
    }

    @Override
    protected void doGet(HttpServletRequest request,
                         HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("application/json");
        response.setCharacterEncoding("UTF-8");

        PrintWriter out = response.getWriter();

        String input = request.getParameter("number");

        try {

            if (input == null || input.trim().isEmpty()) {
                response.setStatus(HttpServletResponse.SC_BAD_REQUEST);
                out.print("{\"error\":\"Please enter a number.\"}");
                return;
            }

            int number = Integer.parseInt(input.trim());

            boolean result = isPrime(number);

            String message;

            if (result) {
                message = number + " is a Prime Number.";
            } else {
                message = number + " is NOT a Prime Number.";
            }

            out.print("{\"number\":" + number +
                    ",\"isPrime\":" + result +
                    ",\"message\":\"" + message + "\"}");

        } catch (NumberFormatException e) {

            response.setStatus(HttpServletResponse.SC_BAD_REQUEST);

            out.print("{\"error\":\"Please enter a valid integer.\"}");
        }
    }
}

    async function checkPrime() {

        const number = document.getElementById("number").value;

        const resultDiv = document.getElementById("result");

        if (number.trim() === "") {
            showResult("Please enter a number.", "error");
            return;
        }

        resultDiv.style.display = "block";
        resultDiv.className = "";
        resultDiv.innerText = "Checking...";

        try {

            const response = await fetch(
                "<%= request.getContextPath() %>/check-prime?number="
                + encodeURIComponent(number)
            );

            const data = await response.json();

            if (!response.ok) {
                showResult(data.error, "error");
                return;
            }

            if (data.isPrime) {

                showResult(data.message, "prime");

            } else {

                showResult(data.message, "not-prime");

            }

        } catch (error) {

            showResult("Something went wrong. Please try again.", "error");

            console.error(error);
        }
    }

    function showResult(message, className) {

        const resultDiv = document.getElementById("result");

        resultDiv.innerText = message;

        resultDiv.className = className;

        resultDiv.style.display = "block";
    }

</script>
